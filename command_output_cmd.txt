# Line format (read by run_cmd.sh):
#   == TEXT            stage heading
#   # TEXT             note shown on screen (lines before the first heading, like these, are skipped)
#   blank              ignored
#   COMMAND            any other line: a shell command, run in the scratch directory,
#                      then a <space> pause
#
# Rubric ids go at the end of the note (or heading) above the commands they label,
# e.g. "# get returns product 1   [A1-03, A1-14]". In this file they come from the
# group it covers (A1-01 to A1-06), and every success-sequence step also carries A1-14.

== 0. Setup
# A scratch directory holding a copy of products.py.
pwd
# no products.json exists yet   [A1-14]
ls

== 1. Required success sequence
# the eight commands from the handout, in order, starting with no products.json   [A1-06, A1-14]
python3 products.py --help
# the two adds assign ids 1 and 2   [A1-01, A1-14]
python3 products.py add --name "Notebook" --price 4.5 --quantity 10
python3 products.py add --name "Pencil" --price 1.25 --quantity 5
# list shows both products, sorted by id   [A1-02, A1-14]
python3 products.py list
# get returns product 1   [A1-03, A1-14]
python3 products.py get 1
# update only the supplied field: name and price are preserved   [A1-04, A1-14]
python3 products.py update 1 --quantity 12
# delete with --yes removes product 2 and returns deleted_id   [A1-05, A1-14]
python3 products.py delete 2 --yes
# final list: only Notebook, price "4.50", quantity 12   [A1-02, A1-14]
python3 products.py list

== 2. Expected final data
# the saved JSON file: only Notebook remains and next_id stays 3   [A1-14]
cat products.json

== 3. A later invocation reads the saved data
# a brand-new python3 process loads products.json from disk   [A1-03, A1-14]
python3 products.py get 1
